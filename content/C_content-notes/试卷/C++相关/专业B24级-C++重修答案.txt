1、题目：使用指针实现一维数组逆序
要求：编写函数void reverseArray(int *arr, int n)，将长度为n的数组逆序，主函数输入输出验证。
#include <stdio.h>
void reverseArray(int *arr, int n) {
    int *start = arr, *end = arr + n - 1;
    while (start < end) {
        int temp = *start;
        *start = *end;
        *end = temp;
        start++;
        end--;
    }
}
int main() {
    int arr[] = {1,2,3,4,5};
    int n = sizeof(arr)/sizeof(arr[0]);
    reverseArray(arr, n);
    for (int i=0; i<n; i++) printf("%d ", arr[i]); // 输出5 4 3 2 1
    return 0;
}
评分标准：程序结构完整 3分、变量定义正确 3分、数据输入正确3分、计算正确3分、输出格式正确3分。

2、题目：计算结构体数组中学生平均分
要求：定义struct Student（含学号、姓名、成绩），输入 3 名学生数据，计算平均分并输出。
#include <stdio.h>
typedef struct {
    int id;
    char name[20];
    float score;
} Student;
int main() {
    Student stu[3];
    float sum = 0;
    for (int i=0; i<3; i++) {
        scanf("%d %s %f", &stu[i].id, stu[i].name, &stu[i].score);
        sum += stu[i].score;
    }
    printf("平均分：%.2f", sum/3);
    return 0;
}
评分标准：程序结构完整 3分、变量定义正确 3分、数据输入正确3分、计算正确3分、输出格式正确3分。
