#include <stdio.h>
void matrixMultiply(int a[][100], int b[][100], int c[][100], int m, int n, int p) {
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < p; j++) {
            c[i][j] = 0;

3. 程序输出结果是什么？power 函数实现了什么算法来计算什么？
#include <stdio.h>
int power(int base, int exponent) {
    if (exponent == 0) {
        return 1;
    } else if (exponent % 2 == 0) {
        int temp = power(base, exponent / 2);
        return temp * temp;
    } else {
        return base * power(base, exponent - 1);
    }
}
int main() {
    int base = 2, exponent = 5;
    printf("%d\n", power(base, exponent));
    return 0;
}
4. 程序输出结果是什么？longes函数实现了什么功能？
#include <stdio.h>
int longes (int arr[], int n) {
    int dp[n];
    for (int i = 0; i < n; i++) {
        dp[i] = 1;
        for (int j = 0; j < i; j++) {
            if (arr[j] < arr[i]) {
                dp[i] = (dp[i] > dp[j] + 1)? dp[i] : dp[j] + 1;
            }
        }
    }
    int maxLen = 0;
    for (int i = 0; i < n; i++) {
        if (dp[i] > maxLen) {
        maxLen = dp[i];
        }
    }
    return maxLen;
}
int main() {
    int a[] = {10, 9, 2, 5, 3, 7, 101, 18};
    int n = sizeof(a) / sizeof(a[0]);
    printf("%d\n", longes (a, n));
    return 0;
}
