1、题目：动态数组实现斐波那契数列
要求：用malloc动态分配数组，存储前 n 项斐波那契数列（n 由用户输入）。
#include <stdio.h>
#include <stdlib.h>
int main() {
    int n;
    scanf("%d", &n);
    int *fib = (int*)malloc(n * sizeof(int));
    fib[0] = 0; fib[1] = 1;
    for (int i=2; i<n; i++) fib[i] = fib[i-1] + fib[i-2];
    for (int i=0; i<n; i++) printf("%d ", fib[i]);
    free(fib);
    return 0;
}
评分标准：程序结构完整 3分、变量定义正确 3分、数据输入正确3分、计算正确3分、输出格式正确3分。

2、题目：文件读写 —— 统计文本文件单词数
要求：读取test.txt，统计单词数量（单词以空格分隔）。
#include <stdio.h>
#include <ctype.h>
int main() {
    FILE *fp = fopen("test.txt", "r");
    if (!fp) return 1;
    int count = 0, in_word = 0;
    char c;
    while ((c = fgetc(fp)) != EOF) {
        if (isspace(c)) {
            in_word = 0;
        } else if (!in_word) {
            in_word = 1;
            count++;
        }
    }
    printf("单词数：%d", count);
    fclose(fp);
    return 0;
}
评分标准：程序结构完整 3分、变量定义正确 3分、数据输入正确3分、计算正确3分、输出格式正确3分。
