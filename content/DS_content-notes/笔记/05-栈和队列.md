# 栈和队列

---

## 一、栈 (Stack)

### 定义

栈是**仅限于在表尾进行插入或删除操作**的受限线性表。

> 普通线性表可以在任意位置进行插入或删除操作，但栈只能在线性表的**表尾**进行。

- **LIFO**：后进先出（Last In First Out）
- **栈顶 (Top)**：进行插入或删除的那一端
- **栈底 (Bottom)**：另一端

### 栈的基本操作

| 操作 | 说明 |
|------|------|
| `InitStack()` | 创建一个栈 |
| `DestroyStack()` | 销毁一个栈（没有栈了） |
| `ClearStack()` | 清空一个栈（栈里面没有元素了） |
| `StackisEmpty()` | 判断一个栈是否为空 |
| `Stacklength()` | 计算一个栈的长度 |
| `push()` | 入栈 |
| `pop()` | 出栈 |
| `Gettop()` | 获取栈顶元素 |

## 顺序栈

### 数据类型定义

```c
struct stack
{
    int data[1024]; // 创建的栈可以存储 1024 个 int 类型的元素
    int top;        // 用来表示栈顶，初始化为 -1
    int length;     // 栈中元素的个数
    int maxlen;     // 创建的栈的长度
};
```

### 头文件 stack.h

```c
#include <stdio.h>
#include <stdlib.h>
#ifndef __STACK_H__
#define __STACK_H__

struct stack
{
    int data[1024];
    int top;
    int length;
    int maxlen;
};

struct stack *InitStack();
void DestroyStack(struct stack *l);
void ClearStack(struct stack *l);
int StackisEmpty(struct stack *l);
int Stacklength(struct stack *l);
void push(struct stack *l, int x);
void pop(struct stack *l, int *p);
int Gettop(struct stack *l);
void printf_stack(struct stack *l);

#endif
```

### 实现 stack.c

```c
#include "stack.h"

struct stack *InitStack(void)
{
    struct stack *l = (struct stack *)malloc(sizeof(struct stack));
    l->top = -1;
    l->length = 0;
    l->maxlen = 1024;
    return l;
}

void DestroyStack(struct stack *l)
{
    if (l == NULL)
    {
        return;
    }
    if (l->top == -1 || l->length == 0)
    {
        free(l);
    }
}

void ClearStack(struct stack *l)
{
    if (l->top == -1 && l->length == 0)
    {
        return;
    }
    l->top = -1;
    l->length = 0;
}

void push(struct stack *l, int x)
{
    if (l->top == l->maxlen - 1)
    {
        return;  // 栈满
    }
    l->data[++l->top] = x;
    ++l->length;
}

void pop(struct stack *l, int *p)
{
    if (l->top == -1 || l->length == 0)
    {
        return;  // 栈空
    }
    *p = l->data[l->top--];
    --(l->length);
}

void printf_stack(struct stack *l)
{
    if (l->top == -1 && l->length == 0)
    {
        return;
    }
    int d = l->top;
    while (d != -1)
    {
        printf("%d ", l->data[d--]);
    }
    printf("\n");
}
```

## 链式栈

用带头结点的双向链表模拟栈。

### 数据类型定义

```c
// 数据节点
struct node
{
    int data;
    struct node *next;  // 指向下一个节点
    struct node *prev;  // 指向上一个节点
};

// 头结点
struct h_node
{
    struct node *first; // 指向栈底
    struct node *top;   // 指向栈顶
    int length;         // 栈中数据节点的个数
};
```

### 头文件 linkedstack.h

```c
#include <stdio.h>
#include <stdlib.h>

#ifndef __LINKEDSTACK_H__
#define __LINKEDSTACK_H__

struct h_node
{
    struct node *first;
    struct node *top;
    int length;
};

struct node
{
    int data;
    struct node *next;
    struct node *prev;
};

struct h_node *Create_Linkedstack();        // 创建一个链式栈
void push(struct h_node *l, int x);         // 入栈
void pop(struct h_node *l);                  // 出栈
void Clear_linkedstack(struct h_node *l);    // 清空栈
void Destroy_linkedstack(struct h_node *l);  // 销毁栈
int linkedstackisempty(struct h_node *l);    // 判断栈是否为空
void printf_node(struct h_node *l);          // 遍历打印栈

#endif
```

### 实现 linkedstack.c

```c
#include "linkedstack.h"

// 创建一个链式栈的头结点
struct h_node *Create_Linkedstack()
{
    struct h_node *l = (struct h_node *)malloc(sizeof(struct h_node));
    l->first = NULL;
    l->top = NULL;
    l->length = 0;
    return l;
}

// 入栈（尾插法）
void push(struct h_node *l, int x)
{
    if (l == NULL) return;
    struct node *p = (struct node *)malloc(sizeof(struct node));
    p->data = x;
    p->next = NULL;
    p->prev = NULL;
    if (l->top == NULL)
    {
        l->top = p;
        l->first = p;
        l->length++;
    }
    else
    {
        l->top->next = p;
        p->prev = l->top;
        l->top = p;
        l->length++;
    }
}

// 出栈
void pop(struct h_node *l)
{
    if (l->top == NULL || l == NULL) return;

    if (l->top == l->first)  // 只有一个节点
    {
        free(l->top);
        l->top = NULL;
        l->first = NULL;
        l->length--;
        return;
    }

    l->top = l->top->prev;
    l->top->next->prev = NULL;
    free(l->top->next);
    l->top->next = NULL;
    l->length--;
}

// 清空栈
void Clear_linkedstack(struct h_node *l)
{
    if (l->top == NULL) return;
    while (l->top != NULL)
    {
        pop(l);
    }
}

// 销毁栈
void Destroy_linkedstack(struct h_node *l)
{
    if (l == NULL) return;
    Clear_linkedstack(l);
    free(l);
}

// 判断栈是否为空
int linkedstackisempty(struct h_node *l)
{
    if (l->length == 0) return 1;
    return 0;
}

// 打印栈（从栈顶到栈底）
void printf_node(struct h_node *l)
{
    if (l->top == NULL) return;
    struct node *p = l->top;
    while (p != NULL)
    {
        printf("%d ", p->data);
        p = p->prev;
    }
    printf("\n");
}
```

## 二、队列 (Queue)

### 定义

队列也是一种**受限的线性表**：允许在一端进行插入操作，另一端进行删除操作。

- **FIFO**：先进先出（First In First Out）
- **队尾 (Rear)**：允许插入元素的一端
- **队头 (Front)**：允许删除元素的一端

### 队列的基本操作

| 操作 | 说明 |
|------|------|
| `InitQueue` | 创建一个队列 |
| `DestroyQueue` | 销毁一个队列 |
| `ClearQueue` | 清空一个队列 |
| `QueueisEmpty` | 判断队列是否为空 |
| `Queuelength` | 求队列长度 |
| `EnQueue` | 进队 |
| `OutQueue` | 出队 |
| `Gethead` | 获取队头元素 |

---

## 顺序队列

### 数据类型定义

```c
#define MAXLEN 1024

struct queue
{
    int data[MAXLEN]; // 队列的数据存储
    int front;        // 队头位置（一维数组的下标）
    int rear;         // 队尾位置（一维数组的下标）
    int length;       // 队列中元素的个数
};
```

### 实现

```c
// 创建队列
struct queue *InitQueue(void)
{
    struct queue *p = (struct queue *)malloc(sizeof(struct queue));
    p->front = -1;
    p->rear = -1;
    p->length = 0;
    return p;
}

// 元素进队
void Enqueue(struct queue *p, int x)
{
    if (p->rear == MAXLEN - 1) return;  // 队列满
    if (p->front == -1)
    {
        p->front = 0;
    }
    p->data[++p->rear] = x;
    ++p->length;
}

// 元素出队
void Outqueue(struct queue *p)
{
    if (p->front == -1 || p->front > p->rear) return;
    ++p->front;
    --p->length;
}

// 清空队列
void Clearqueue(struct queue *p)
{
    if (p->front == -1 && p->rear == -1) return;
    p->front = -1;
    p->rear = -1;
    p->length = 0;
}

// 销毁队列
void Destroyqueue(struct queue *p)
{
    if (p == NULL) return;
    free(p);
}

// 队列长度
int Queuelength(struct queue *p)
{
    if (p == NULL) return 0;
    return p->length;
}
```

## 链式队列

用带头结点的双向链表实现。

### 数据类型定义

```c
// 数据节点
struct node
{
    int data;
    struct node *next;
    struct node *prev;
};

// 头结点
struct h_node
{
    struct node *front; // 指向队头节点
    struct node *rear;  // 指向队尾节点
    int length;         // 数据节点个数
};
```

### 实现

```c
// 创建队列
struct h_node *Creat_queue(void)
{
    struct h_node *p = (struct h_node *)malloc(sizeof(struct h_node));
    p->front = NULL;
    p->rear = NULL;
    p->length = 0;
    return p;
}

// 元素入队（尾插法）
void Enqueue(struct h_node *l, int x)
{
    if (l == NULL) return;
    struct node *p = (struct node *)malloc(sizeof(struct node));
    p->data = x;
    p->next = NULL;
    p->prev = NULL;
    if (l->rear == NULL)
    {
        l->front = p;
        l->rear = p;
        ++l->length;
    }
    else
    {
        l->rear->next = p;
        p->prev = l->rear;
        l->rear = p;
        ++l->length;
    }
}

// 元素出队（删除第一个节点）
void Outqueue(struct h_node *l)
{
    if (l == NULL) return;
    if (l->front == l->rear && l->length == 1)  // 只有一个元素
    {
        free(l->front);
        l->front = NULL;
        l->rear = NULL;
        l->length = 0;
    }
    else
    {
        l->front = l->front->next;
        l->front->prev->next = NULL;
        free(l->front->prev);
        l->front->prev = NULL;
        --l->length;
    }
}

// 清空队列
void Clearqueue(struct h_node *l)
{
    if (l == NULL) return;
    while (l->front != NULL)
    {
        Outqueue(l);
    }
}

// 销毁队列
void Destroyqueue(struct h_node *l)
{
    if (l == NULL) return;
    Clearqueue(l);
    free(l);
}

// 判断队列是否为空
int QueueisEmpty(struct h_node *l)
{
    if (l == NULL) return 1;
    if (l->front == NULL && l->rear == NULL && l->length == 0) return 1;
    return 0;
}
```

## 应用：用栈求表达式值（中缀表达式求值）

### 算法概述

使用两个栈（**操作数栈** 和 **运算符栈**）对中缀表达式进行求值，核心思想是利用运算符优先级来控制计算顺序。

### 运算符优先级表

| 运算符 | 栈内优先级(isp) | 栈外优先级(icp) |
|--------|:------:|:------:|
| `+` `-` | 3 | 3 |
| `*` `/` | 5 | 5 |
| `(` | 1 | 6 |
| `)` | 6 | 1 |
| `#`（结束符） | 0 | 0 |

### 算法步骤

1. 在表达式末尾添加结束符 `#`，并将 `#` 压入运算符栈
2. 从左到右扫描表达式：
   - **操作数** → 直接压入操作数栈
   - **运算符** → 比较当前运算符（icp）与栈顶运算符（isp）：
     - icp > isp → 当前运算符入栈，继续扫描下一个字符
     - icp < isp → 弹出栈顶运算符，从操作数栈弹出两个操作数计算，结果入操作数栈
     - icp == isp → 弹出栈顶的 `(`（遇到 `)` 时），继续扫描

### 示例：`5+30*4+7/3`

扫描过程逐步分析：

| 步骤 | 当前字符 | 操作数栈 | 运算符栈 | 说明 |
|:----:|:--------:|:---------:|:---------:|:-----|
| 1 | `5` | `5` | `#` | 操作数入栈 |
| 2 | `+` | `5` | `# +` | `+` icp(3) > `#` isp(0)，入栈 |
| 3 | `30` | `5, 30` | `# +` | 操作数入栈 |
| 4 | `*` | `5, 30` | `# + *` | `*` icp(5) > `+` isp(3)，入栈 |
| 5 | `4` | `5, 30, 4` | `# + *` | 操作数入栈 |
| 6 | `+` | `5, 120` | `# +` | `+` icp(3) < `*` isp(5)，弹出 `*`，计算 `30*4=120`，结果入栈；继续比较 `+` icp(3) == `+` isp(3)，弹出 `+`，计算 `5+120=125` |
| 7 | — | `125` | `#` | `+` icp(3) > `#` isp(0)，入栈 |
| 8 | `7` | `125, 7` | `# +` | 操作数入栈 |
| 9 | `/` | `125, 7` | `# + /` | `/` icp(5) > `+` isp(3)，入栈 |
| 10 | `3` | `125, 7, 3` | `# + /` | 操作数入栈 |
| 11 | `#` | `125, 2` | `# +` | `#` icp(0) < `/` isp(5)，弹出 `/`，计算 `7/3=2`（整数除法），结果入栈 |
| 12 | — | `127` | `#` | `#` icp(0) < `+` isp(3)，弹出 `+`，计算 `125+2=127`，结果入栈 |
| 13 | — | `127` | `#` | `#` icp(0) == `#` isp(0)，算法结束 |

**最终结果：127**

### 代码实现

```c
#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

#define STACK_SIZE 64

/* 运算符栈 */
struct op_stack {
    char data[STACK_SIZE];
    int top;
};

/* 操作数栈 */
struct num_stack {
    int data[STACK_SIZE];
    int top;
};

void op_push(struct op_stack *s, char c) {
    s->data[++s->top] = c;
}

char op_pop(struct op_stack *s) {
    return s->data[s->top--];
}

char op_top(struct op_stack *s) {
    return s->data[s->top];
}

void num_push(struct num_stack *s, int v) {
    s->data[++s->top] = v;
}

int num_pop(struct num_stack *s) {
    return s->data[s->top--];
}

/* 栈内优先级 (isp) */
int isp(char c) {
    switch (c) {
        case '#': return 0;
        case '(': return 1;
        case '+': case '-': return 3;
        case '*': case '/': return 5;
        case ')': return 6;
        default: return -1;
    }
}

/* 栈外优先级 (icp) */
int icp(char c) {
    switch (c) {
        case '#': return 0;
        case ')': return 1;
        case '+': case '-': return 3;
        case '*': case '/': return 5;
        case '(': return 6;
        default: return -1;
    }
}

/* 双目运算 */
int calc(int a, char op, int b) {
    switch (op) {
        case '+': return a + b;
        case '-': return a - b;
        case '*': return a * b;
        case '/': return b != 0 ? a / b : 0;
        default: return 0;
    }
}

int evaluate(const char *expr) {
    struct op_stack ops = {{'#'}, 0};   // 以 '#' 初始化运算符栈
    struct num_stack nums = {{0}, -1};  // 操作数栈
    int i = 0;

    while (expr[i] != '#' || op_top(&ops) != '#') {
        if (expr[i] == '#' && op_top(&ops) == '#')
            break;

        if (isdigit(expr[i])) {
            /* 解析多位整数 */
            int val = 0;
            while (isdigit(expr[i])) {
                val = val * 10 + (expr[i] - '0');
                i++;
            }
            num_push(&nums, val);
        } else {
            /* 运算符 */
            if (icp(expr[i]) > isp(op_top(&ops))) {
                op_push(&ops, expr[i]);
                i++;
            } else if (icp(expr[i]) < isp(op_top(&ops))) {
                char op = op_pop(&ops);
                int b = num_pop(&nums);
                int a = num_pop(&nums);
                num_push(&nums, calc(a, op, b));
            } else {
                /* 相等：弹出栈顶运算符（处理括号匹配） */
                op_pop(&ops);
                i++;
            }
        }
    }
    return num_pop(&nums);
}

int main() {
    printf("5+30*4+7/3 = %d\n", evaluate("5+30*4+7/3#"));
    return 0;
}
```

## 栈在嵌入式系统中的应用

### 函数调用栈 (Call Stack)

MCU（如 ARM Cortex-M 系列）上电后，SP（Stack Pointer）寄存器指向栈顶。每次函数调用时：

1. **调用者保存现场**：将返回地址（LR）、局部变量、寄存器值压栈
2. **被调用者分配空间**：为局部变量在栈上分配内存
3. **函数返回时恢复现场**：出栈还原寄存器值，PC 跳回返回地址

```
高地址
  ┌──────────────┐  ← MSP (主栈指针)
  │  局部变量     │
  ├──────────────┤
  │  返回地址(LR) │
  ├──────────────┤
  │  寄存器(R4-R11)│
  ├──────────────┤
  │  参数传递区   │
  └──────────────┘  ← 栈底
低地址
```

**栈溢出风险**：嵌入式系统中栈空间通常有限（几 KB 到几十 KB），深度递归或过大的局部数组容易导致栈溢出，造成 HardFault。

### 中断栈 (Interrupt Stack)

ARM Cortex-M 内核提供 **MSP（主栈指针）** 和 **PSP（进程栈指针）**：

- **MSP**：用于中断服务函数（ISR）和内核代码
- **PSP**：用于任务/线程代码

中断发生时硬件自动压栈 8 个寄存器（xPSR、PC、LR、R12、R3-R0），中断返回时硬件自动出栈恢复。这种硬件栈操作保证了中断响应时间的确定性。

### RTOS 任务栈 (Task Stack)

FreeRTOS / RT-Thread 等 RTOS 为每个任务分配独立的栈空间：

```c
// FreeRTOS 创建任务时分配栈
#define TASK_STACK_SIZE 256  // 256 个 Word（约 1KB）

TaskHandle_t task_handle;
xTaskCreate(
    vTaskFunction,      // 任务函数
    "Task1",            // 任务名
    TASK_STACK_SIZE,    // 栈深度（单位：Word）
    NULL,               // 参数
    1,                  // 优先级
    &task_handle        // 句柄
);
```

任务切换（Context Switch）时，PendSV 中断将当前任务的寄存器压入其私有栈，再从下一个任务的栈中弹出寄存器值，实现任务切换。

**栈深度测量**：RTOS 通常提供 `uxTaskGetStackHighWaterMark()` 等 API 来检测任务栈的剩余空间，帮助开发者优化栈大小。

## 三、循环队列 (Circular Queue)

### 为什么需要循环队列

顺序队列存在 **"假溢出"** 问题：随着不断出队，`front` 指针后移，队头前的空间被浪费，而 `rear` 已到达数组末尾时无法再入队，但数组前半部分实际是空的。

循环队列通过将数组逻辑上视为一个环，将 `front` 和 `rear` 在到达末尾时回绕到开头，从而复用已出队的空间，解决了假溢出问题。

### 循环队列 vs 顺序队列

| 对比项 | 顺序队列 | 循环队列 |
|--------|:--------:|:--------:|
| 空间利用率 | 低（假溢出） | 高（可复用） |
| 实现复杂度 | 简单 | 稍复杂 |
| 嵌入式场景 | 极少使用 | 广泛使用 |

### 数据类型定义

```c
#define MAX_SIZE 64

struct circular_queue {
    int data[MAX_SIZE];
    int front;  // 队头下标
    int rear;   // 队尾下标（指向下一个空位）
};
```

### 核心操作实现

```c
/* 初始化 */
void init_queue(struct circular_queue *q) {
    q->front = 0;
    q->rear = 0;
}

/* 判断队空 */
int is_empty(struct circular_queue *q) {
    return q->front == q->rear;
}

/* 判断队满（牺牲一个元素空间来判满） */
int is_full(struct circular_queue *q) {
    return (q->rear + 1) % MAX_SIZE == q->front;
}

/* 入队 */
int enqueue(struct circular_queue *q, int val) {
    if (is_full(q)) return -1;  // 队满
    q->data[q->rear] = val;
    q->rear = (q->rear + 1) % MAX_SIZE;
    return 0;
}

/* 出队 */
int dequeue(struct circular_queue *q, int *val) {
    if (is_empty(q)) return -1;  // 队空
    *val = q->data[q->front];
    q->front = (q->front + 1) % MAX_SIZE;
    return 0;
}

/* 队列长度 */
int queue_length(struct circular_queue *q) {
    return (q->rear - q->front + MAX_SIZE) % MAX_SIZE;
}
```

#### UART 环形缓冲区

串口通信中最常用的数据结构，配合中断使用：

```c
#define UART_BUF_SIZE 256

static uint8_t uart_rx_buf[UART_BUF_SIZE];

        rx_head = next;
    } /* 满则丢弃（溢出处理） */
}

uint8_t uart_read_byte(uint8_t *byte) {
    if (rx_tail == rx_head) return 0;  // 空
    *byte = uart_rx_buf[rx_tail];
    rx_tail = (rx_tail + 1) % UART_BUF_SIZE;
    return 1;
}
```

## 四、单调栈与单调队列

### 单调栈 (Monotonic Stack)

栈内元素保持单调递增或单调递减。常用于解决 **Next Greater Element（下一个更大元素）** 类问题。

**典型场景**：
- 找出数组中每个元素右边第一个比它大的元素（O(n)）
- 接雨水（LeetCode 42）
- 柱状图中最大的矩形（LeetCode 84）

```c
// 下一个更大元素：单调递减栈
int* nextGreaterElement(int* nums, int numsSize, int* returnSize) {
    int *res = (int*)malloc(numsSize * sizeof(int));
    int stack[numsSize], top = -1;
    for (int i = 0; i < numsSize; i++) res[i] = -1;

    for (int i = 0; i < numsSize; i++) {
        while (top >= 0 && nums[stack[top]] < nums[i]) {
            res[stack[top]] = nums[i];
            top--;
        }
        stack[++top] = i;
    }
    *returnSize = numsSize;
    return res;
}
```

### 单调队列 (Monotonic Queue)

队列内元素保持单调性，常用于维护滑动窗口内的最值。

**典型场景**：
- 滑动窗口最大值（LeetCode 239）
- 和至少为 K 的最短子数组（LeetCode 862）

单调队列通常用双端队列（deque）实现，队首始终是当前窗口的最值。详见"面试练习题"中滑动窗口最大值问题。

### 1. 用两个栈实现队列

**题目**：用两个栈实现一个队列，支持 `push(int x)`（队尾入队）和 `pop()`（队头出队）操作。

**虚拟来源/年份**：虚拟互联网公司甲 2021

**思路**：
- 使用两个栈 `s1` 和 `s2`。`s1` 负责入队，`s2` 负责出队。
- `push` 时直接将元素压入 `s1`。
- `pop` 时，若 `s2` 为空，则将 `s1` 中所有元素依次弹出并压入 `s2`（此时 `s2` 栈顶即为队头元素），然后从 `s2` 弹出；若 `s2` 非空，直接从 `s2` 弹出。

**摊销分析**：每个元素最多入栈两次（`s1` 入、`s1` 进 `s2`）、出栈两次（`s1` 出、`s2` 出），均摊时间复杂度 O(1)。

**代码实现**：

```c
typedef struct {
    int *s1;
    int top1;
    int *s2;
    int top2;
    int capacity;
} MyQueue;

MyQueue* myQueueCreate() {
    MyQueue *q = (MyQueue *)malloc(sizeof(MyQueue));
    q->capacity = 1024;
    q->s1 = (int *)malloc(q->capacity * sizeof(int));
    q->s2 = (int *)malloc(q->capacity * sizeof(int));
    q->top1 = -1;
    q->top2 = -1;
    return q;
}

void myQueuePush(MyQueue *obj, int x) {
    obj->s1[++obj->top1] = x;
}

int myQueuePop(MyQueue *obj) {
    if (obj->top2 == -1) {
        while (obj->top1 >= 0) {
            obj->s2[++obj->top2] = obj->s1[obj->top1--];
        }
    }
    return obj->s2[obj->top2--];
}

int myQueuePeek(MyQueue *obj) {
    if (obj->top2 == -1) {
        while (obj->top1 >= 0) {
            obj->s2[++obj->top2] = obj->s1[obj->top1--];
        }
    }
    return obj->s2[obj->top2];
}

int myQueueEmpty(MyQueue *obj) {
    return obj->top1 == -1 && obj->top2 == -1;
}

void myQueueFree(MyQueue *obj) {
    free(obj->s1);
    free(obj->s2);
    free(obj);
}
```

### 2. 用两个队列实现栈

**题目**：用两个队列实现一个栈，支持 `push(int x)`（入栈）和 `pop()`（出栈）操作。

**虚拟来源/年份**：虚拟互联网公司丙 2020

**思路**：
- 使用两个队列 `q1` 和 `q2`。
- `push` 时：元素入队到非空的那个队列（初始都为空时入 `q1`）。
- `pop` 时：将非空队列中除最后一个元素外的所有元素出队并转移到另一个队列，然后弹出最后一个元素（即栈顶）。

**代码实现**：

```c
typedef struct {
    int data[1024];
    int front;
    int rear;
    int size;
} Queue;

void q_init(Queue *q) { q->front = 0; q->rear = 0; q->size = 0; }
void q_push(Queue *q, int x) { q->data[q->rear++] = x; q->size++; }
int q_pop(Queue *q) { q->front++; q->size--; return q->data[q->front - 1]; }
int q_empty(Queue *q) { return q->size == 0; }

typedef struct {
    Queue q1, q2;
} MyStack;

MyStack* myStackCreate() {
    MyStack *s = (MyStack *)malloc(sizeof(MyStack));
    q_init(&s->q1);
    q_init(&s->q2);
    return s;
}

void myStackPush(MyStack *obj, int x) {
    if (!q_empty(&obj->q1))
        q_push(&obj->q1, x);
    else
        q_push(&obj->q2, x);
}

int myStackPop(MyStack *obj) {
    Queue *non_empty = &obj->q1;
    Queue *empty = &obj->q2;
    if (q_empty(non_empty)) {
        non_empty = &obj->q2;
        empty = &obj->q1;
    }
    while (non_empty->size > 1) {
        q_push(empty, q_pop(non_empty));
    }
    return q_pop(non_empty);
}
```

### 3. 有效的括号

**题目**：给定一个只包含 `'('`, `')'`, `'{'`, `'}'`, `'['`, `']'` 的字符串，判断字符串是否有效。有效需满足：左括号必须用相同类型的右括号闭合；左括号必须以正确的顺序闭合。

**虚拟来源/年份**：虚拟互联网公司丙 2022

**思路**：
- 遍历字符串，遇到左括号时压入栈。
- 遇到右括号时，检查栈顶是否为对应的左括号——是则弹出，否则无效。
- 遍历结束后栈为空则有效。

**代码实现**：

```c
#include <stdbool.h>
#include <string.h>

bool isValid(char *s) {
    int len = strlen(s);
    char stack[len];
    int top = -1;

    for (int i = 0; i < len; i++) {
        if (s[i] == '(' || s[i] == '{' || s[i] == '[') {
            stack[++top] = s[i];
        } else {
            if (top == -1) return false;
            char left = stack[top--];
            if ((s[i] == ')' && left != '(') ||
                (s[i] == '}' && left != '{') ||
                (s[i] == ']' && left != '[')) {
                return false;
            }
        }
    }
    return top == -1;
}
```

### 4. 最小栈（O(1) 获取最小值）

**题目**：设计一个栈，支持 `push`、`pop`、`top` 操作，且能在 O(1) 时间内获取栈中最小元素。

**虚拟来源/年份**：虚拟互联网公司甲 2022

**思路**：
- 使用两个栈：数据栈 `data` 和最小值栈 `min`。
- `push(x)`：`data` 正常入栈；若 `min` 为空或 `x <= min` 栈顶，则 `x` 也入 `min` 栈。
- `pop()`：若 `data` 出栈的元素等于 `min` 栈顶，则 `min` 也出栈。
- `getMin()`：返回 `min` 栈顶元素。

**代码实现**：

```c
typedef struct {
    int data[1024];
    int top_data;
    int min_stack[1024];
    int top_min;
} MinStack;

MinStack* minStackCreate() {
    MinStack *s = (MinStack *)malloc(sizeof(MinStack));
    s->top_data = -1;
    s->top_min = -1;
    return s;
}

void minStackPush(MinStack *obj, int x) {
    obj->data[++obj->top_data] = x;
    if (obj->top_min == -1 || x <= obj->min_stack[obj->top_min]) {
        obj->min_stack[++obj->top_min] = x;
    }
}

void minStackPop(MinStack *obj) {
    if (obj->data[obj->top_data] == obj->min_stack[obj->top_min]) {
        obj->top_min--;
    }
    obj->top_data--;
}

int minStackTop(MinStack *obj) {
    return obj->data[obj->top_data];
}

int minStackGetMin(MinStack *obj) {
    return obj->min_stack[obj->top_min];
}

void minStackFree(MinStack *obj) {
    free(obj);
}
```

**优化思路**：若空间敏感，可在 `min` 栈中存储差值而非完整值，或使用一个栈存储 `(值, 当前最小值)` 的结构体。

### 5. 表达式求值（中缀表达式）

**题目**：实现一个计算器，计算包含 `+`、`-`、`*`、`/` 和括号的中缀表达式。

**虚拟来源/年份**：虚拟通信公司己 2021

**思路**：
- 使用两个栈：操作数栈和运算符栈。
- 扫描表达式，通过运算符优先级决定是否立即计算。
- 遇到 `(` 直接入运算符栈，遇到 `)` 则计算直到弹出 `(`。
- 优先级规则：`*` `/` > `+` `-`。
- 详见本文"应用：用栈求表达式值"章节中的完整实现和优先级表。

**更简洁的通用实现（支持括号）**：

```c
#include <stdio.h>
#include <ctype.h>
#include <string.h>

int calc(int a, char op, int b) {
    switch (op) {
        case '+': return a + b;
        case '-': return a - b;
        case '*': return a * b;
        case '/': return b ? a / b : 0;
        default:  return 0;
    }
}

int priority(char op) {
    if (op == '+' || op == '-') return 1;
    if (op == '*' || op == '/') return 2;
    return 0;
}

int evaluate(char *expr) {
    int nums[64], ntop = -1;
    char ops[64]; int otop = -1;
    int i = 0;

    while (expr[i]) {
        if (isdigit(expr[i])) {
            int val = 0;
            while (isdigit(expr[i])) val = val * 10 + (expr[i++] - '0');
            nums[++ntop] = val;
        } else if (expr[i] == '(') {
            ops[++otop] = expr[i++];
        } else if (expr[i] == ')') {
            while (otop >= 0 && ops[otop] != '(') {
                int b = nums[ntop--], a = nums[ntop--];
                nums[++ntop] = calc(a, ops[otop--], b);
            }
            otop--; i++;
        } else {
            while (otop >= 0 && priority(ops[otop]) >= priority(expr[i])) {
                int b = nums[ntop--], a = nums[ntop--];
                nums[++ntop] = calc(a, ops[otop--], b);
            }
            ops[++otop] = expr[i++];
        }
    }
    while (otop >= 0) {
        int b = nums[ntop--], a = nums[ntop--];
        nums[++ntop] = calc(a, ops[otop--], b);
    }
    return nums[ntop];
}
```

### 6. 用栈实现括号匹配

**题目**：给定一个包含 `'('`、`')'`、`'{'`、`'}'`、`'['`、`']'` 的字符串，判断括号是否匹配。

**虚拟来源/年份**：虚拟电商公司乙 2020

**思路**：此题与第 3 题"有效的括号"相同，但需要额外考虑字符串中可能包含其他字符（仅需处理括号）。

**扩展要求**：有时面试官会要求同时检查括号的**嵌套深度**，此时可用一个计数器记录当前栈的深度。

```c
int maxDepth(char *s) {
    int max = 0, cur = 0;
    int len = strlen(s);
    char stack[len];
    int top = -1;

    for (int i = 0; s[i]; i++) {
        if (s[i] == '(' || s[i] == '[' || s[i] == '{') {
            stack[++top] = s[i];
            cur = top + 1;
            if (cur > max) max = cur;
        } else if (s[i] == ')' || s[i] == ']' || s[i] == '}') {
            if (top == -1) return -1;
            char left = stack[top];
            if ((s[i] == ')' && left != '(') ||
                (s[i] == ']' && left != '[') ||
                (s[i] == '}' && left != '{')) {
                return -1;
            }
            top--;
        }
    }
    return top == -1 ? max : -1;
}
```

### 7. 滑动窗口最大值（单调队列）

**题目**：给定一个整数数组 `nums` 和一个大小为 `k` 的滑动窗口，窗口从数组最左端移动到最右端，输出每个窗口中的最大值。

**虚拟来源/年份**：虚拟互联网公司甲 2022

**思路**：
- 使用**双端队列**（deque）存储数组元素的下标，保持队列中下标对应的值**单调递减**。
- 窗口右移时：
  1. 队首元素若已不在窗口内（下标 < 窗口左边界），则出队。
  2. 新元素入队时，从队尾移除所有比它小的元素（这些元素永远不可能是最大值）。
  3. 当前窗口最大值即队首元素对应的值。

**时间复杂度**：O(n)，每个元素最多入队一次、出队一次。

**代码实现**：

```c
/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* maxSlidingWindow(int* nums, int numsSize, int k, int* returnSize) {
    *returnSize = numsSize - k + 1;
    int *res = (int *)malloc(*returnSize * sizeof(int));

    int deque[numsSize];
    int head = 0, tail = -1;
    int idx = 0;

    for (int i = 0; i < numsSize; i++) {
        if (head <= tail && deque[head] < i - k + 1) {
            head++;
        }
        while (head <= tail && nums[deque[tail]] <= nums[i]) {
            tail--;
        }
        deque[++tail] = i;
        if (i >= k - 1) {
            res[idx++] = nums[deque[head]];
        }
    }
    return res;
}
```

### 8. 循环队列的设计

**题目**：设计一个循环队列，支持 `enQueue`（入队）、`deQueue`（出队）、`Front`（获取队头元素）、`Rear`（获取队尾元素）、`isEmpty`、`isFull` 操作。

**虚拟来源/年份**：虚拟生活服务公司戊 2021

**思路**：
- 使用固定大小数组，通过 `front` 和 `rear` 指针配合取模运算实现循环。
- 判空：`front == rear`
- 判满：`(rear + 1) % capacity == front`（牺牲一个空间判满，或单独用 `size` 变量记录长度）
- 入队时 `rear` 后移取模，出队时 `front` 后移取模。

**代码实现（使用 size 变量判满，不浪费空间）**：

```c
typedef struct {
    int *data;
    int front;
    int rear;
    int size;
    int capacity;
} MyCircularQueue;

MyCircularQueue* myCircularQueueCreate(int k) {
    MyCircularQueue *q = (MyCircularQueue *)malloc(sizeof(MyCircularQueue));
    q->data = (int *)malloc(k * sizeof(int));
    q->front = 0;
    q->rear = 0;
    q->size = 0;
    q->capacity = k;
    return q;
}

int myCircularQueueEnQueue(MyCircularQueue *obj, int value) {
    if (obj->size == obj->capacity) return 0;
    obj->data[obj->rear] = value;
    obj->rear = (obj->rear + 1) % obj->capacity;
    obj->size++;
    return 1;
}

int myCircularQueueDeQueue(MyCircularQueue *obj) {
    if (obj->size == 0) return 0;
    obj->front = (obj->front + 1) % obj->capacity;
    obj->size--;
    return 1;
}

int myCircularQueueFront(MyCircularQueue *obj) {
    if (obj->size == 0) return -1;
    return obj->data[obj->front];
}

int myCircularQueueRear(MyCircularQueue *obj) {
    if (obj->size == 0) return -1;
    int tail = (obj->rear - 1 + obj->capacity) % obj->capacity;
    return obj->data[tail];
}

int myCircularQueueIsEmpty(MyCircularQueue *obj) {
    return obj->size == 0;
}

int myCircularQueueIsFull(MyCircularQueue *obj) {
    return obj->size == obj->capacity;
}

void myCircularQueueFree(MyCircularQueue *obj) {
    free(obj->data);
    free(obj);
}
```

### 9. 栈的压入弹出序列判断

**题目**：输入两个整数序列，第一个序列表示栈的压入顺序，判断第二个序列是否为该栈的弹出顺序。假设入栈的所有数字均不相等。

**虚拟来源/年份**：虚拟通信公司己 2021

**思路**：
- 模拟入栈出栈过程：使用一个辅助栈，按照压入序列的顺序入栈。
- 每次入栈后，检查栈顶是否等于弹出序列的当前元素——若相等则弹出，并继续检查下一个弹出元素。
- 最终若辅助栈为空，则弹出序列合法。

**示例**：`pushSeq = [1,2,3,4,5]`，`popSeq = [4,5,3,2,1]` → true

| 步骤 | 入栈 | 出栈检查 | 辅助栈 |
|:----:|:----:|:---------:|:------:|
| 1 | 1 | 栈顶 1 != 4 | [1] |
| 2 | 2 | 栈顶 2 != 4 | [1,2] |
| 3 | 3 | 栈顶 3 != 4 | [1,2,3] |
| 4 | 4 | 栈顶 4 == 4，弹出 | [1,2,3] |
| 5 | — | 栈顶 3 != 5 | [1,2,3] |
| 6 | 5 | 栈顶 5 == 5，弹出 | [1,2,3] |
| 7 | — | 栈顶 3 == 3，弹出 | [1,2] |
| 8 | — | 栈顶 2 == 2，弹出 | [1] |
| 9 | — | 栈顶 1 == 1，弹出 | [] |

成功，所有元素入栈出栈完毕，最终栈空 → true。

**代码实现**：

```c
#include <stdbool.h>

bool validateStackSequences(int* pushed, int pushedSize,
                             int* popped, int poppedSize) {
    int stack[pushedSize];
    int top = -1;
    int popIdx = 0;

    for (int i = 0; i < pushedSize; i++) {
        stack[++top] = pushed[i];
        while (top >= 0 && stack[top] == popped[popIdx]) {
            top--;
            popIdx++;
        }
    }
    return top == -1;
}
```
