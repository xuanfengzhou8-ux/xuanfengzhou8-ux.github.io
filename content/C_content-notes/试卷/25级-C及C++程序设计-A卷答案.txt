五、编程题（每题15分，共30分）
1、编写函数 isprime (int a)，判断整型数 a 是否为素数，若是返回 1，否则返回 0，在主函数中调用该函数并输出结果。
#include <stdio.h>
#include <math.h>
int isprime(int a) {
    if (a <= 1) return 0;
    for (int i = 2; i <= sqrt(a); i++) {
        if (a % i == 0) return 0;
    }
    return 1;
}
int main() {
    int num;
    printf("请输入一个整数：");
    scanf("%d", &num);
    if (isprime(num)) {
        printf("%d是素数\n", num);
    } else {
        printf("%d不是素数\n", num);
    }
    return 0;
}
评分标准：程序结构完整 3分、变量定义正确 3分、数据输入正确3分、计算正确3分、输出格式正确3分。

2、题目：计算结构体数组中学生平均分
要求：定义struct Student（含学号、姓名、成绩），输入 3 名学生数据，计算平均分并输出。
#include <stdio.h>
typedef struct {
    int id;
    char name[20];
    float score;
} Student;
int main() {
    Student stu[3];
    float sum = 0;
    for (int i=0; i<3; i++) {
        scanf("%d %s %f", &stu[i].id, stu[i].name, &stu[i].score);
        sum += stu[i].score;
    }
    printf("平均分：%.2f", sum/3);
    return 0;
}
评分标准：程序结构完整 3分、变量定义正确 3分、数据输入正确3分、计算正确3分、输出格式正确3分。
