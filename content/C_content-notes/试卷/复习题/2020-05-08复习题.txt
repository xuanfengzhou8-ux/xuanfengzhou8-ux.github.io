2、
struct stu
{ int x ;
int *y;
} *p ;
int dt[4]={ 10 , 20 , 30 , 40 };
struct stu a[4]={50 , &dt[0] , 60 , &dt[1] , 70 , &dt[2] , 80 , &dt[3] };
main()
{ p=a;
  printf("%d," , ++p->x); //语句1
  printf("%d," , (++p)->x ); //语句2
  printf("%d\n" , ++(*p->y) ); //语句3
}

A.10 , 20 , 20 B.50 , 60 , 21 C.51 , 60 , 21 D 60 , 70 , 31
