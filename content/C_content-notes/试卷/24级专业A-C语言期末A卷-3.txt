#include <iostream>
#include <stdlib.h>
using namespace std;
struct TT{
    int id;struct TT *next;
};                // 定义结构体 2分
struct TT * cjlb(){  // 创建链表 5分
    struct TT *p,*h,*Q;
    int N;
    Q=h=p=(struct TT *)malloc(sizeof(struct TT));
    cin >> N;
    if (N!=-1){
        h->id=N;h->next=NULL;
    }
    while(1){
        cin >> N;
        if (N==-1)
            break;
        p=(struct TT *)malloc(sizeof(struct TT));
        p->id=N;Q->next=p;Q=p;p->next=NULL;
    }
    return h;
}
struct TT * DEL(struct TT *h,int n=45){ // 删除节点 5分
    struct TT *Q,*W;	W=Q=h;
    while(Q){
        if(Q->id==n){
            W->next=Q->next;	free(Q);break;
        }
        W=Q;Q=Q->next;
    }
    return h;
}
void PP(struct TT *h){ // 输出链表 3分
    while(h != NULL)
        cout << h->id<<endl;h=h->next;
}
int main(){
    struct TT *h;   // 定义结构体指针变量  2分
    h=cjlb();      //调用创建链表函数 1分
    int M;
    cout<<"请输入需要删除的数"<<endl; cin>>N;//输入删除的节点  1分
    h=DEL(h,M);	//调用创建链表函数 1分

    return 0;
}
