# 单链表

## 基本概念

### 数据节点定义

```c
struct node
{
    int data;           // 数据域：用来保存数据
    struct node *next;  // 指针域：用来保存逻辑关系（指向下一个节点）
};
```

### 静态建立链表

保存 (1, 5, 4, 3, 2)：

```c
#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

int main()
{
    struct node a1, a2, a3, a4, a5;
    struct node *p = &a1;

    // 保存数据
    a1.data = 1;
    a2.data = 5;
    a3.data = 4;
    a4.data = 3;
    a5.data = 2;

    // 保存关系
    a1.next = &a2;
    a2.next = &a3;
    a3.next = &a4;
    a4.next = &a5;
    a5.next = NULL;

    // 遍历打印
    while (p != NULL)
    {
        printf("%d ", p->data);
        p = p->next;
    }
    printf("\n");
}
```

### 动态建立链表

```c
int d;
while(1)
{
    scanf("%d", &d);
    if(d == 0)
    {
        break;
    }
    struct node *p = (struct node *)malloc(sizeof(struct node));
    p->data = d;
    p->next = NULL;
}
```

## 单链表的主要操作：增删改查

### 头文件 Linklist.h

```c
#include <stdio.h>
#include <stdlib.h>
#ifndef __LINKLIST_H__
#define __LINKLIST_H__

struct node
{
    int data;          // 数据域
    struct node *next; // 指针域
};

// 头结点
struct h_node
{
    struct node *first; // 整个链表的第一个节点
    struct node *last;  // 整个链表的最后一个节点
};

struct h_node *Great_h_node(void);      // 创建一个头结点
struct node *Great_node(void);          // 创建一个链表（增）
struct node *Great_increasing(void);    // 插入排序创建增序链表
struct node *delect_x(struct node *h, int x);  // 删除节点（删）
struct node *change_x(struct node *h, int x);  // 修改节点（改）
void find_x(struct node *h, int x);     // 查找节点（查）
void printf_linklist(struct node *h);   // 打印链表
void destroy_linklist(struct node *h);  // 销毁链表

#endif
```

### 增 — 创建一个链表

**尾插法**（打印顺序 = 输入顺序）与**头插法**（打印顺序 = 输入逆序）：

```c
/*
 * Great_node:
 *   创建一个链表
 *   返回值: 创建链表的第一个节点的地址
 */
struct node *Great_node(void)
{
    int d;                     // 用来保存数据域
    struct node *last = NULL;  // 用来保存最后一个节点
    struct node *first = NULL; // 用来保存链表的第一个节点

    while (1)
    {
        scanf("%d", &d);
        if (d == 0)
        {
            break;
        }
        struct node *p = (struct node *)malloc(sizeof(struct node));
        p->data = d;
        p->next = NULL;
        if (first == NULL)
        {
            first = p;
            last = p;
        }
        else
        {
            // 尾插法:
            // last->next = p;
            // last = p;

            // 头插法:
            p->next = first;
            first = p;
        }
    }
    return first;
}
```

### 删 — 删除链表中 x 对应的节点

```c
/*
 * delect_x:
 *   删除链表中 x 对应的那个节点
 *   @h: 链表的第一个节点
 *   @x: 要删除的数据
 *   返回值: 返回链表的首节点
 */
struct node *delect_x(struct node *h, int x)
{
    struct node *pr = NULL; // 指向找到的节点的上一个节点
    struct node *px = h;    // 用来找要删除的节点

    while (px != NULL)
    {
        if (px->data == x)
        {
            break;  // 找到了！
        }
        pr = px;
        px = px->next;
    }

    if (px == h)  // 删除的是第一个节点
    {
        h = h->next;
        px->next = NULL;
        free(px);
    }
    else           // 删除的是中间或最后节点
    {
        pr->next = px->next;
        px->next = NULL;
        free(px);
    }
    return h;
}
```

### 查 — 在链表中查找元素

```c
/*
 * find_x:
 *   遍历链表找到 x
 *   @h: 链表的首节点
 *   @x: 要查找的元素值
 */
void find_x(struct node *h, int x)
{
    if (h == NULL)
    {
        return;
    }
    while (h != NULL)
    {
        if (h->data == x)
        {
            printf("%d\n", h->data);
            return;
        }
        h = h->next;
    }
    printf("没找到\n");
}
```

### 打印链表

```c
void printf_linklist(struct node *h)
{
    if (h == NULL)
    {
        return;
    }
    while (h != NULL)
    {
        printf("%d ", h->data);

```

### 销毁链表

```c
void destroy_linklist(struct node *h)
{
    if (h == NULL)
    {
        return;
    }
    struct node *px = h;    // px 用来遍历链表的每一个节点
    struct node *pr = NULL; // pr 用来做事（干掉一个节点）
    while (px != NULL)
    {
        pr = px;
        px = px->next;
        pr->next = NULL;
        free(pr);
    }
}
```

## 插入排序 — 创建增序链表

```c
/*
 * Great_increasing:
 *   插入排序创建一个增序链表
 *   返回值: 返回创建的链表的第一个节点
 */
struct node *Great_increasing(void)
{
    int d;
    struct node *last = NULL;  // 用来保存最后一个节点
    struct node *first = NULL; // 用来保存链表的第一个节点
    struct node *pr = NULL;    // 插入点前面的那个节点
    struct node *px = NULL;    // 保存插入点的位置

    while (1)
    {
        scanf("%d", &d);
        if (d == 0)
        {
            break;
        }
        struct node *p = (struct node *)malloc(sizeof(struct node));
        p->data = d;
        p->next = NULL;

        if (first == NULL) // p 是第一个节点
        {
            first = p;
            last = p;
        }
        else
        {
            px = first;
            while (px != NULL)
            {
                if (px->data > d) // 找到第一个比 d 大的节点
                {
                    break;
                }
                pr = px;
                px = px->next;
            }

            if (px == NULL)      // 没找到 → 尾插法
            {
                pr->next = p;
                last = p;
            }
            else if (px == first) // 第一个节点就是插入位置 → 头插法
            {
                p->next = first;
                first = p;
            }
            else                  // 中间插入
            {
                p->next = px;
                pr->next = p;
            }
        }
    }
    return first;
}
```

## 快慢指针技术

快慢指针（也称龟兔赛跑算法）是链表问题中一种非常经典的双指针技巧。其核心思想是：使用两个指针同时遍历链表，但移动速度不同——**慢指针每次走一步，快指针每次走两步**。

### 典型应用

1. **寻找链表中间节点**：快指针到达末尾时，慢指针恰好在中间。
2. **检测链表是否有环**：如果存在环，快慢指针最终会在环内相遇。
3. **寻找倒数第 k 个节点**：让快指针先走 k 步，然后快慢指针同步前进，快指针走到末尾时慢指针即为倒数第 k 个节点。

### 快慢指针的 C 语言实现框架

```c
/*
 * 快慢指针基本框架
 * 返回链表中间节点
 */
struct node *find_middle(struct node *head)
{
    if (head == NULL)
    {
        return NULL;
    }

    struct node *slow = head;
    struct node *fast = head;

    while (fast != NULL && fast->next != NULL)
    {
        slow = slow->next;       // 慢指针走一步
        fast = fast->next->next; // 快指针走两步
    }

    // 当链表节点个数为奇数时，slow 恰好指向中间节点
    // 当链表节点个数为偶数时，slow 指向中间靠右的节点
    return slow;
}
```

> **适用场景**：当题目要求"只能遍历一次"或"O(1) 额外空间"时，应优先考虑快慢指针。

## 常见陷阱与注意事项

在嵌入式开发中操作单链表时，以下问题需要特别关注：

### 1. 空指针解引用（Null Pointer Dereference）

不断访问 `p->next` 之前，必须确保 `p` 不为 NULL。此问题常出现在链表的末尾操作中：

```c
// 错误写法 — 访问了 NULL 的 next 成员
while (p->next != NULL) { p = p->next; }

// 正确写法
while (p != NULL && p->next != NULL) { p = p->next; }
```

### 2. 内存泄漏（Memory Leak）

删除节点时，必须先保存下一个节点的地址，再释放当前节点，否则会丢失后续节点的引用：

```c
// 错误写法 — 先 free 再访问 next，导致内存泄漏
struct node *p = head;
while (p != NULL)
{
    free(p);          // p 已释放，但 p->next 无法获取
    p = p->next;      // 已访问非法内存
}

// 正确写法
struct node *p = head;
while (p != NULL)
{
    struct node *tmp = p->next;
    free(p);
    p = tmp;
}
```

### 3. 悬挂指针（Dangling Pointer）

释放节点后，若未将相关指针置 NULL，后续可能出现悬挂引用：

```c
free(px);
px = NULL;           // 好的习惯：释放后将指针置 NULL
// 若还有其他指针指向该节点，也应一并处理
```

### 4. 头节点变更问题

在头插、删除头节点等操作中，链表的头指针可能发生变化，函数必须通过返回值或二级指针将新的头节点地址传回调用者：

```c
// 推荐：通过返回值返回新头节点
struct node *delect_x(struct node *h, int x);

// 或者：使用二级指针
void delect_x(struct node **h, int x);
```

## 链表 vs 数组：嵌入式系统中的选型分析

嵌入式系统资源受限（RAM 小、无 MMU、无操作系统或仅有 RTOS），在链表与数组之间的选择需要综合权衡：

| 维度 | 数组 | 链表 |
|------|------|------|
| 随机访问 | O(1)，通过下标直接访问 | O(n)，必须遍历 |
| 插入/删除（非末尾） | O(n)，需要移动大量元素 | O(1)，只需修改指针 |
| 空间开销 | 无额外开销 | 每个节点额外占用 4/8 字节指针空间 |
| 内存连续性 | 连续内存 | 离散内存 |
| 缓存局部性 | 优秀（缓存预取友好） | 差（跳跃访问，缓存命中率低） |

### 嵌入式环境下的关键考量

**1. 内存碎片问题**

数组需要一块连续的内存块。在长时间运行的嵌入式系统中，反复 malloc/free 会导致堆内存碎片化，此时可能无法分配足够大的连续数组，而链表可以容忍碎片化，每个节点只需分配小块内存。

**2. 动态分配约束**

许多嵌入式环境禁止或限制动态内存分配（MISRA-C 规范要求尽量避免 malloc/free），因为：
- malloc 可能失败（返回 NULL），需要额外的错误处理
- 实时系统中 malloc 的执行时间不确定
- 内存碎片不可预测

此时更倾向于使用**静态数组**或**内存池（Memory Pool）** + 静态链表。

**3. 实时性要求**

数组的遍历时间可预测（连续内存，无指针跳转），而链表的遍历因节点分散在内存各处，访存时间波动较大。在硬实时（Hard Real-Time）系统中，数组通常更安全。

**4. 静态链表（嵌入式常用方案）**

在禁止动态分配的嵌入式场景中，可以使用**静态链表**——预先定义一个大的节点数组，通过游标（下标）代替指针来组织链表：

```c
#define MAX_NODES 100

struct static_node
{
    int data;
    int next;  // 使用下标代替指针，-1 表示 NULL
};

struct static_node pool[MAX_NODES];
int head = -1; // 链表头，-1 表示空链表
```

> **总结**：在 RAM 充裕、对实时性要求不苛刻的应用中，链表灵活性更高；在资源极度受限或硬实时系统中，优先考虑数组或静态链表。

## 循环链表与环检测（Floyd 判环算法）

循环链表是单链表的一种变体，其中最后一个节点的 `next` 指针指向链表中的某个节点（而非 NULL），从而形成环状结构。

### Floyd 判环算法（快慢指针法）

Floyd 判环算法是检测链表是否有环的最经典方法，由 Robert W. Floyd 提出，时间复杂度 O(n)，空间复杂度 O(1)。

**核心思想**：
- 慢指针每次走 1 步，快指针每次走 2 步
- 若无环：快指针最终到达 NULL
- 若有环：快慢指针一定在环内相遇

**算法推导**：
设环外长度为 a，环长度为 b，慢指针入环时快指针已在环内距离入环点 x 步。设两指针相遇时慢指针在环内走了 s 步，则快指针走了 2s 步。由于在环内快指针比慢指针多走了若干整圈，有 `2s - s = n * b`（n 为整数），即 `s = n * b`。因此相遇点距离入环点的步数为 `s % b`。若让一个指针从头开始，另一个从相遇点开始，都以一步前进，则它们会在入环点相遇。

```c
/*
 * has_cycle:
 *   检测链表是否有环（Floyd 判环算法）
 *   @head: 链表头节点
 *   返回值: 1 表示有环，0 表示无环
 */
int has_cycle(struct node *head)
{
    if (head == NULL || head->next == NULL)
    {
        return 0;
    }

    struct node *slow = head;
    struct node *fast = head;

    while (fast != NULL && fast->next != NULL)
    {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast)
        {
            return 1;  // 相遇即有环
        }
    }
    return 0;  // 快指针到末尾，无环
}

/*
 * detect_cycle_entry:
 *   返回环的入口节点（若存在环）
 *   @head: 链表头节点
 *   返回值: 环入口节点地址，无环则返回 NULL
 */
struct node *detect_cycle_entry(struct node *head)
{
    struct node *slow = head;
    struct node *fast = head;
    int has_cycle = 0;

    // 第一阶段：判断是否有环
    while (fast != NULL && fast->next != NULL)
    {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast)
        {
            has_cycle = 1;
            break;
        }
    }

    if (!has_cycle)
    {
        return NULL;
    }

    // 第二阶段：找入环点
    // 将一个指针重置到头部，两指针同步前进，相遇点即入环点
    slow = head;
    while (slow != fast)
    {
        slow = slow->next;
        fast = fast->next;
    }
    return slow;
}
```

## 面试练习题

### 反转链表（Reverse Linked List）

**虚拟互联网公司丙 2022 校招 / 虚拟电商公司乙 2021 春招**

**题目**：给定单链表的头节点 `head`，反转链表并返回反转后的链表头。

**解题思路**：
反转链表是链表题中最基础的考点。核心思路是**迭代地改变指针方向**：遍历链表时，将当前节点的 `next` 指向前一个节点。由于单向链表无法回溯到前驱节点，需要三个指针协同工作：

- `prev`：指向已完成反转部分的头节点（当前节点的前驱）
- `curr`：指向当前正在处理的节点
- `next`：保存当前节点的原始下一个节点（防止断链）

**时间复杂度**：O(n)，只需遍历一次链表
**空间复杂度**：O(1)，只使用了常数个指针变量

```c
struct node *reverse_list(struct node *head)
{
    struct node *prev = NULL;
    struct node *curr = head;
    struct node *next = NULL;

    while (curr != NULL)
    {
        next = curr->next;   // 保存下一个节点（防止断链）
        curr->next = prev;   // 反转当前节点的指针方向
        prev = curr;         // 前驱指针前移
        curr = next;         // 当前指针前移
    }
    return prev;  // prev 指向原链表的末尾，即新链表的头
}
```

**进阶 — 递归实现**（理解递归思想对面试有帮助）：

```c
struct node *reverse_list_recursive(struct node *head)
{
    if (head == NULL || head->next == NULL)
    {
        return head;  // 递归出口：空链表或只有一个节点
    }

    struct node *new_head = reverse_list_recursive(head->next);
    head->next->next = head;  // 将下一个节点指回当前节点
    head->next = NULL;        // 断开当前节点原有的 next

    return new_head;
}
```

### 链表中环的检测（Cycle Detection）

**虚拟互联网公司甲 2021 秋招 / 虚拟电商公司乙 2023 实习**

**题目**：判断一个单链表是否有环，并找到环的入口节点。

**解题思路**：
使用 Floyd 判环算法。维护两个指针——快指针每次走两步，慢指针每次走一步：

1. 若快指针遇到 NULL，说明无环
2. 若快慢指针相遇，说明有环
3. 找入环点时，将慢指针重置到链表头，快慢指针都改为每次走一步，再次相遇的位置即为入环点

**数学原理**：
设环外长度为 a，环长为 b。两指针第一次相遇时，慢指针走了 `a + s` 步，快指针走了 `a + s + n*b` 步（n 为快指针绕的圈数）。由快指针走的是慢指针的两倍得：`2(a + s) = a + s + n*b`，即 `a + s = n*b`，所以 `a = (n-1)*b + (b - s)`。这意味着从链表头到入环点的距离，等于从相遇点继续前进 `(n-1)圈 + (b-s)` 的步长，通过同步移动即可找到入环点。

具体代码已在上方"Floyd 判环算法"一节给出，此处不再重复。

**面试追问**：如果要求在找到环后**计算环的长度**，该怎么做？
答：在快慢指针相遇后，让一个指针停在该位置，另一个指针继续绕环走，再次相遇时走过的步数即为环长。

### 合并两个有序链表

**虚拟生活服务公司戊 2022 春招 / 虚拟互联网公司甲 2021 秋招**

**题目**：输入两个单调递增的链表，合并为一个递增链表后返回。

**解题思路**：
使用**双指针** + **哨兵节点（dummy node）** 技巧。同时遍历两链表，每次取出值较小的节点接到结果链表后面。哨兵节点可以避免单独处理头节点的特殊情况。

```c
/*
 * merge_sorted_lists:
 *   合并两个有序递增链表
 *   @l1: 第一个有序链表头
 *   @l2: 第二个有序链表头
 *   返回值: 合并后的有序链表头
 */
struct node *merge_sorted_lists(struct node *l1, struct node *l2)
{
    // 处理空链表情况
    if (l1 == NULL) return l2;
    if (l2 == NULL) return l1;

    // 哨兵节点，简化头节点的处理
    struct node dummy;
    struct node *tail = &dummy;
    dummy.next = NULL;

    while (l1 != NULL && l2 != NULL)
    {
        if (l1->data <= l2->data)
        {
            tail->next = l1;
            l1 = l1->next;
        }
        else
        {
            tail->next = l2;
            l2 = l2->next;
        }
        tail = tail->next;
    }

    // 拼接剩余部分
    if (l1 != NULL)
    {
        tail->next = l1;
    }
    else
    {
        tail->next = l2;
    }

    return dummy.next;
}
```

**递归解法**（思路更简洁，但函数调用栈深度为 O(n)）：

```c
struct node *merge_sorted_lists_recursive(struct node *l1, struct node *l2)
{
    if (l1 == NULL) return l2;
    if (l2 == NULL) return l1;

    if (l1->data <= l2->data)
    {
        l1->next = merge_sorted_lists_recursive(l1->next, l2);
        return l1;
    }
    else
    {
        l2->next = merge_sorted_lists_recursive(l1, l2->next);
        return l2;
    }
}
```

### 删除链表倒数第 N 个节点

**虚拟通信公司己 2021 秋招 / 虚拟搜索公司丁 2022 校招**

**题目**：给定一个链表，删除链表的倒数第 n 个节点并返回链表头（n 保证有效）。

**解题思路**：
使用**双指针（快慢指针）** 技巧，只需一次遍历：

1. 让快指针先走 n 步
2. 然后快慢指针同时前进，直到快指针到达链表末尾
3. 此时慢指针指向的正是倒数第 n 个节点的前驱节点
4. 删除慢指针的下一个节点即可

```c
/*
 * remove_nth_from_end:
 *   删除链表倒数第 n 个节点
 *   @head: 链表头节点
 *   @n: 倒数位置（从 1 开始计数）
 *   返回值: 新链表头节点
 */
struct node *remove_nth_from_end(struct node *head, int n)
{
    if (head == NULL)
    {
        return NULL;
    }

    // 哨兵节点，方便处理删除头节点的情况
    struct node dummy;
    dummy.next = head;

    struct node *fast = &dummy;
    struct node *slow = &dummy;

    // 快指针先走 n 步
    for (int i = 0; i < n && fast != NULL; i++)
    {
        fast = fast->next;
    }

    // 快慢指针同步前进，直到快指针到末尾
    while (fast->next != NULL)
    {
        slow = slow->next;
        fast = fast->next;
    }

    // 此时 slow 指向待删节点的前驱
    struct node *to_delete = slow->next;
    slow->next = to_delete->next;
    to_delete->next = NULL;
    free(to_delete);

    return dummy.next;
}
```

**注意事项**：
- 使用哨兵节点（dummy node）统一处理删除头节点的边界情况
- n 的有效性校验（题目一般保证有效，但工程中应当做检查）
- 释放被删除节点的内存

### 链表的中间节点（快慢指针）

**虚拟门户公司辛 2020 秋招 / 虚拟生活服务公司戊 2023 春招**

**题目**：找到单链表的中间节点。若链表节点数为奇数，返回中间节点；若为偶数，返回中间两个节点中的第二个（靠右的那个）。

**解题思路**：
经典快慢指针应用。快指针每次走两步，慢指针每次走一步。当快指针到达末尾时，慢指针正好在中间。

```c
/*
 * find_middle_node:
 *   返回链表的中间节点
 *   @head: 链表头节点
 *   返回值: 中间节点地址
 */
struct node *find_middle_node(struct node *head)
{
    if (head == NULL)
    {
        return NULL;
    }

    struct node *slow = head;
    struct node *fast = head;

    while (fast != NULL && fast->next != NULL)
    {
        slow = slow->next;       // 慢指针走一步
        fast = fast->next->next; // 快指针走两步
    }

    return slow;
}
```

**不同偶数情况下的变体**：
- 要求返回靠左的节点：将循环条件改为 `fast->next != NULL && fast->next->next != NULL`
- 上述两种变体的区别在于快指针的边界判断条件不同

### 回文链表（Palindrome Linked List）

**虚拟互联网公司甲 2022 春招 / 虚拟互联网公司丙 2023 校招**

**题目**：判断一个链表是否为回文链表（正序和逆序遍历值相同）。

**解题思路**：
三步法（空间复杂度 O(1)）：

1. **找中间节点**：使用快慢指针找到链表的中点
2. **反转后半部分**：从中间节点之后开始，反转链表后半部分
3. **前后比较**：将前半部分与反转后的后半部分逐一比较，全部相同即为回文

最后应将链表恢复原状（面试中可说明，但非必须）。

```c
/*
 * is_palindrome:
 *   判断链表是否为回文链表
 *   @head: 链表头节点
 *   返回值: 1 表示是回文，0 表示不是
 */
int is_palindrome(struct node *head)
{
    if (head == NULL || head->next == NULL)
    {
        return 1;  // 空链表或只有一个节点，视为回文
    }

    // 步骤 1：快慢指针找中间节点
    struct node *slow = head;
    struct node *fast = head;
    while (fast != NULL && fast->next != NULL)
    {
        slow = slow->next;
        fast = fast->next->next;
    }

    // 步骤 2：反转后半部分链表
    // 若节点数为奇数，slow 指向正中间节点，应将其跳过（正中间不影响回文判断）
    struct node *second_half = slow;
    if (fast != NULL)  // 节点数为奇数
    {
        second_half = slow->next;
    }

    // 反转后半部分
    struct node *prev = NULL;
    struct node *curr = second_half;
    struct node *next = NULL;
    while (curr != NULL)
    {
        next = curr->next;
        curr->next = prev;
        prev = curr;
        curr = next;
    }
    struct node *reversed_second = prev;  // 反转后的后半部分头

    // 步骤 3：比较前后半段
    struct node *first = head;
    struct node *second = reversed_second;
    while (second != NULL)
    {
        if (first->data != second->data)
        {
            return 0;  // 发现不匹配
        }
        first = first->next;
        second = second->next;
    }

    return 1;
}
```

**时间复杂度**：O(n)，**空间复杂度**：O(1)

**提示**：也可以用数组或栈来解（将链表值复制到数组后用双指针判断），但空间复杂度为 O(n)，面试中应优先给出 O(1) 空间的解法。

### 删除链表中的重复元素

**虚拟零售公司庚 2021 秋招 / 虚拟互联网公司甲 2022 实习**

**题目**：在一个排序的链表中，删除所有重复的节点。有两种变形：
- **变体 A（保留一个）**：每个重复元素只保留一个
- **变体 B（全部删除）**：重复的节点一个不留

#### 变体 A — 保留一个

```c
/*
 * delete_duplicates_keep_one:
 *   删除排序链表中的重复元素，每个值只保留一个
 *   @head: 链表头节点
 *   返回值: 新链表头节点
 */
struct node *delete_duplicates_keep_one(struct node *head)
{
    struct node *curr = head;

    while (curr != NULL && curr->next != NULL)
    {
        if (curr->data == curr->next->data)
        {
            struct node *dup = curr->next;
            curr->next = dup->next;
            dup->next = NULL;
            free(dup);
        }
        else
        {
            curr = curr->next;  // 不重复才前移
        }
    }
    return head;
}
```

#### 变体 B — 全部删除（不保留任何重复值）

```c
/*
 * delete_duplicates_remove_all:
 *   删除排序链表中所有重复的节点，重复值一个不留
 *   @head: 链表头节点
 *   返回值: 新链表头节点
 */
struct node *delete_duplicates_remove_all(struct node *head)
{
    if (head == NULL)
    {
        return NULL;
    }

    struct node dummy;       // 哨兵节点，方便处理头节点被删除的情况
    dummy.next = head;

    struct node *prev = &dummy;
    struct node *curr = head;

    while (curr != NULL)
    {
        // 检测当前节点是否有重复
        int is_dup = 0;
        while (curr->next != NULL && curr->data == curr->next->data)
        {
            // 跳过所有值与 curr 相同的节点
            struct node *tmp = curr->next;
            curr->next = tmp->next;
            tmp->next = NULL;
            free(tmp);
            is_dup = 1;
        }

        if (is_dup)
        {
            // 当前 curr 本身也是重复值，需要删除
            struct node *tmp = curr;
            prev->next = curr->next;
            curr = curr->next;
            tmp->next = NULL;
            free(tmp);
        }
        else
        {
            prev = curr;
            curr = curr->next;
        }
    }

    return dummy.next;
}
```

**时间复杂度**：O(n)，**空间复杂度**：O(1)

**注意**：上述代码假设链表是排序的（升序）。如果未排序，则需要先排序或使用哈希表记录出现次数。

### 链表相交问题

**虚拟互联网公司丙 2022 秋招 / 虚拟互联网公司甲 2023 春招**

**题目**：给两个单链表，判断它们是否相交（即是否存在某个节点同时属于两个链表），如果相交则返回相交的第一个节点。

**解题思路**：
链表相交是指两个链表从某个节点开始，后续所有节点都相同（形成 Y 形结构），而非 X 形交叉。

**方法一：双指针同步法（推荐，O(n) 时间，O(1) 空间）**
1. 分别遍历两个链表，计算它们的长度 `lenA` 和 `lenB`
2. 让较长链表的指针先走 `|lenA - lenB|` 步
3. 然后两个指针同时前进，相遇的第一个节点即为相交点

**方法二：交替遍历法（更巧妙的实现）**
两个指针分别从两个链表头出发，当某个指针走到末尾时，将其重定向到另一个链表的头部。若链表相交，两个指针会在相交节点相遇（走过相同的路程）；若不相交，两个指针最终会同时为 NULL。

```c
/*
 * get_intersection_node:
 *   找到两个链表的相交节点
 *   @headA: 链表 A 的头节点
 *   @headB: 链表 B 的头节点
 *   返回值: 相交节点地址，若不相交则返回 NULL
 */
struct node *get_intersection_node(struct node *headA, struct node *headB)
{
    if (headA == NULL || headB == NULL)
    {
        return NULL;
    }

    // 方法一：长度差法
    int lenA = 0, lenB = 0;
    struct node *pa = headA;
    struct node *pb = headB;

    // 计算长度
    while (pa != NULL) { lenA++; pa = pa->next; }
    while (pb != NULL) { lenB++; pb = pb->next; }

    // 重置指针
    pa = headA;
    pb = headB;

    // 长链表的指针先走差值步
    int diff = lenA - lenB;
    if (diff > 0)  // A 更长
    {
        for (int i = 0; i < diff; i++) pa = pa->next;
    }
    else           // B 更长或等长
    {
        for (int i = 0; i < -diff; i++) pb = pb->next;
    }

    // 同时前进，找第一个相同的节点
    while (pa != NULL && pb != NULL)
    {
        if (pa == pb)
        {
            return pa;  // 找到相交节点
        }
        pa = pa->next;
        pb = pb->next;
    }

    return NULL;  // 不相交
}
```

**方法二 — 交替遍历法**（代码更简洁）：

```c
struct node *get_intersection_node_alternative(struct node *headA, struct node *headB)
{
    if (headA == NULL || headB == NULL)
    {
        return NULL;
    }

    struct node *pa = headA;
    struct node *pb = headB;

    // 当 pa == pb 时退出循环
    // 若相交：相遇在相交节点
    // 若不相交：相遇在 NULL（两个指针都遍历完两个链表后同时为 NULL）
    while (pa != pb)
    {
        pa = (pa == NULL) ? headB : pa->next;
        pb = (pb == NULL) ? headA : pb->next;
    }

    return pa;
}
```

**时间复杂度**：O(m + n)，其中 m、n 分别是两个链表的长度
**空间复杂度**：O(1)

**关键点**：比较的是**节点的地址**（`pa == pb`），而非节点中的数据值。因为相交是物理上的地址相同，而非值相等。
