3、求二维数组中的“鞍点”元素
“鞍点”：是指该元素是所在行的最大值，是所在列的最小值.
参考代码：
#include<stdio.h>
#define N 4
#define M 3
int main()
{
    int a[N][M]={0};
    int i,j,max,t,k,c;
    int flag=0;

    //初始化矩阵
    for(i = 0; i < N; i++)
    {
        for(j = 0; j < M; j++)
            scanf("%d",&a[i][j]);
    }

    for(i = 0;i < N; i++)   //按行查找
    {
        max = a[i][0];    //用来保存最大值
        t = 0;			  //保存最大值的下标
        for(j = 0;j < M;j++)    //找出行的最大值
        {
            if(a[i][j] > max)
            {
                max = a[i][j];
                t = j;
            }
        }

        c = 1;
        for(k = 0;k < N;k++)	//判断此行的最大值是否是此列的最小值
        {
            if(max > a[k][t])
            {
                c = 0;break;	//列中还有比它更小的，所以不是鞍点
            }
        }

        if(c == 1)
        {
            printf("%d\n", max);
            flag++;				//至少有一个鞍点
        }
    }

    if(flag == 0)
        printf("NO");
    printf("\n");

    return 0;
}
