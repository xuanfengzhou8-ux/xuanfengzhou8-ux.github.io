3、写一个函数，求一个一维数组中第二大的元素？
参考代码：
#include <stdio.h>
#define N 10
int main()
{
    int i;
    int a[N] = {0};
    int x,d;

    for(i = 0; i < N; i++)
    {
        scanf("%d",&a[i]);
    }

    if(a[0] > a[1])
    {
        d = a[0];
        x = a[1];
    }
    else
    {
        d = a[1];
        x = a[0];
    }

    for(i = 2; i < N; i++)
    {
        if(a[i] > d)
        {
            x = d;
            d = a[i];
        }
        else
        {
            if(a[i] > x)
            {
                x = a[i];
            }
        }
    }
    printf("%d\n",x);

    for(i = 0; i < N; i++)
    {
        if(a[i] == x)
            printf("%d\n",i);
    }
}
