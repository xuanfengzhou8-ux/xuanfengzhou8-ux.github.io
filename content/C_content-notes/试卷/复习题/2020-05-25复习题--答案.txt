已知两个链表head1和head2 各自有序，请把它们合并成一个链表依然有序。
参考答案：
#include <stdio.h>
#include <stdlib.h>

typedef int ElemType;

typedef struct node
{
    ElemType data;
    struct node * next;
}Node;

typedef struct linkedlist
{
    ElemType nodeNum;
    Node *first;
    struct node *last;
}LinkedList;

LinkedList *create_head()
{
    LinkedList *l = (LinkedList *)malloc(sizeof(*l));
    l->nodeNum = 0;
    l->first = NULL;
    l->last = NULL;

    return l;
}

LinkedList *create_linkedlist()
{
    LinkedList *l = create_head();

    Node *p = NULL;
    int x;

    while(1)
    {
        scanf("%d",&x);
        if(x == 0)
        {
            break;
        }
        //申请一个结点类型的空间把我新输入的x值存到此空间中去
        p = (Node *)malloc(sizeof(*p));
        p->data = x;
        p->next = NULL;

        if(l->first == NULL)	//头为空时
        {
            l->first = p;
            l->last = p;
        }
        else
        {
            Node *pB = NULL;
            Node *pre = NULL;

            pB = l->first;
            while(pB)		//从头开始循环遍历我已有的链表
            {
                if(pB->data > p->data)	//找第一个比我大的结点
                {
                    break;
                }
                pre = pB;			//pre紧跟在pB的后面
                pB = pB->next;
            }
            if(pB == NULL)		//pB找到头了都没有找到比我大的
            {
                l->last->next = p;
                l->last = p;
            }
            else					//有比我大的
            {
                if(pB == l->first)		//第一个就比我大
                {
                    p->next = l->first;
                    l->first = p;
                }
                else					//比我大的在中间
                {
                    p->next = pB;
                    pre->next = p;
                }
            }
        }
    }
    return l;
}

//归并两个链表的函数
LinkedList *merge(LinkedList *A, LinkedList *B)
{
    LinkedList *C = create_head();		//创建一个头结点

    Node *a = NULL;		//指向A链表的数据结点头部
    Node *b = NULL;		//指向B链表的数据结点头部

    //让hc和tc都指向一个空结点
    C->first = (Node *)malloc(sizeof(*C->first));
    C->first->next = NULL;
    C->last = C->first;

    a = A->first;
    b = B->first;

    while(a && b)	//遍历a、b两个链表  当有一个遍历完时结束循环
    {
        if(a->data <= b->data)		//把小的插到新链表的尾部
        {
            C->last->next = a;
            C->last = a;
            a = a->next;
        }
        else
        {
            C->last->next = b;
            C->last = b;
            b = b->next;
        }
    }
    C->last->next = a ? a : b;		//当a、b中的某个链表还有元素是就直接接到新链表的尾部
    C->last = a ? A->last : B->last;	//找到链表的尾结点接起来

    C->first = C->first->next;	//把链表的头结点接起来

    return C;		//返回头结点
}


void print_list(LinkedList *l)
{
    if(l == NULL && l->first == NULL)
    {
        return ;
    }

    Node *p = l->first;
    while(p)
    {
        printf("%d ", p->data);
        p = p->next;
    }
    printf("\n");
}

int main(int argc, char *argv[])
{
    LinkedList *l1 = create_linkedlist();
    LinkedList *l2 = create_linkedlist();
    print_list(l1);
    print_list(l2);

    LinkedList *l3 = merge(l1,l2);
    print_list(l3);

    return 0;
}
